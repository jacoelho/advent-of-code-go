package aoc2023

import (
	"bytes"
	"io"
	"os"
	"testing"
)

func BenchmarkDay17(b *testing.B) {
	actual, err := os.ReadFile("../../inputs/2023/17.txt")
	if err != nil {
		b.Fatal(err)
	}
	sample := []byte(`2413432311323
3215453535623
3255245654254
3446585845452
4546657867536
1438598798454
4457876987766
3637877979653
4654967986887
4564679986453
1224686865563
2546548887735
4322674655533`)
	fixtures := []struct {
		name  string
		input []byte
		want  string
		solve func(io.Reader) (string, error)
	}{
		{"part1/sample", sample, "102", day17p01},
		{"part1/actual", actual, "755", day17p01},
		{"part2/sample", sample, "94", day17p02},
		{"part2/actual", actual, "881", day17p02},
	}
	for _, fixture := range fixtures {
		b.Run(fixture.name, func(b *testing.B) {
			b.ReportAllocs()
			for b.Loop() {
				got, err := fixture.solve(bytes.NewReader(fixture.input))
				if err != nil || got != fixture.want {
					b.Fatalf("solver() = %q, %v; want %q", got, err, fixture.want)
				}
			}
		})
	}
}
