package aoc2019

import (
	"bytes"
	"io"
	"os"
	"testing"
)

func BenchmarkDay18(b *testing.B) {
	actual, err := os.ReadFile("../../inputs/2019/18.txt")
	if err != nil {
		b.Fatal(err)
	}
	part1 := []byte(`#################
#i.G..c...e..H.p#
########.########
#j.A..b...f..D.o#
########@########
#k.E..a...g..B.n#
########.########
#l.F..d...h..C.m#
#################`)
	part2 := []byte(`#############
#g#f.D#..h#l#
#F###e#E###.#
#dCba@#@BcIJ#
#############
#nK.L@#@G...#
#M###N#H###.#
#o#m..#i#jk.#
#############`)
	fixtures := []struct {
		name  string
		input []byte
		want  string
		solve func(io.Reader) (string, error)
	}{
		{"part1/sample", part1, "136", day18p01},
		{"part1/actual", actual, "3862", day18p01},
		{"part2/sample", part2, "72", day18p02},
		{"part2/actual", actual, "1626", day18p02},
	}
	for _, fixture := range fixtures {
		b.Run(fixture.name, func(b *testing.B) {
			b.ReportAllocs()
			for b.Loop() {
				got, err := fixture.solve(bytes.NewReader(fixture.input))
				if err != nil || got != fixture.want {
					b.Fatalf("solver() = %q, %v; want %q", got, err, fixture.want)
				}
			}
		})
	}
}
